cmake_minimum_required(VERSION 3.22)
project(vehlog_portable LANGUAGES C)

add_subdirectory(../components/vl_core vl_core)
add_subdirectory(../components/vl_text vl_text)

option(VEHLOG_BUILD_FREERTOS_ADAPTER "Build the adapter using an existing freertos_kernel target" OFF)
if(VEHLOG_BUILD_FREERTOS_ADAPTER)
    if(NOT TARGET freertos_kernel)
        message(FATAL_ERROR "Add upstream FreeRTOS-Kernel before enabling VEHLOG_BUILD_FREERTOS_ADAPTER")
    endif()
    add_subdirectory(../components/vl_freertos_can vl_freertos_can)
endif()

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(vehlog_core_test ../tests/test_core.c)
        add_executable(vehlog_text_test ../tests/test_text.c)
        target_link_libraries(vehlog_core_test PRIVATE vehlog_core)
        target_link_libraries(vehlog_text_test PRIVATE vehlog_text)

        foreach(test_target IN ITEMS vehlog_core_test vehlog_text_test)
            set_target_properties(${test_target} PROPERTIES
                                  C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
            if(MSVC)
                target_compile_options(${test_target} PRIVATE /UNDEBUG)
            else()
                target_compile_options(${test_target} PRIVATE -UNDEBUG)
            endif()
            add_test(NAME ${test_target} COMMAND ${test_target})
        endforeach()
    endif()
endif()
